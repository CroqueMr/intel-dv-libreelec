cmake_minimum_required(VERSION 3.22)
project(DVBridgeKodiCheck LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
add_compile_options(-UNDEBUG)
get_filename_component(LAB "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(DVBRIDGE_KODI_SOURCE "${LAB}/kodi" CACHE PATH "Patched Kodi source tree")
set(DVBRIDGE_COMMON_SOURCE "${LAB}/common" CACHE PATH "Common DV helpers")
set(RENDERERS "${DVBRIDGE_KODI_SOURCE}/xbmc/cores/VideoPlayer/VideoRenderers")
add_subdirectory("${DVBRIDGE_COMMON_SOURCE}" common)
file(READ "${DVBRIDGE_COMMON_SOURCE}/pack_gles300.vert" DVBRIDGE_VERTEX_SOURCE)
file(READ "${DVBRIDGE_COMMON_SOURCE}/pack_gles300.frag" DVBRIDGE_FRAGMENT_SOURCE)
configure_file("${RENDERERS}/DVBridgeShaders.h.in" DVBridgeShaders.h @ONLY)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DVEgl REQUIRED IMPORTED_TARGET egl)
add_executable(kodi_bridge_smoke "${RENDERERS}/DVBridgeGLES.cpp" ../kodi_bridge_smoke.cpp)
target_include_directories(kodi_bridge_smoke PRIVATE "${DVBRIDGE_KODI_SOURCE}/xbmc" "${RENDERERS}"
                                                    "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(kodi_bridge_smoke PRIVATE dvbridge_core PkgConfig::DVEgl)
enable_testing()
add_executable(playback_policy ../playback_policy.cpp)
target_include_directories(playback_policy PRIVATE "${DVBRIDGE_KODI_SOURCE}/xbmc")
add_test(NAME playback_policy COMMAND playback_policy)
add_executable(dmabuf_layout ../dmabuf_layout.c)
target_link_libraries(dmabuf_layout PRIVATE dvbridge_core)
add_test(NAME dmabuf_layout COMMAND dmabuf_layout)
add_executable(fel_queue ../fel_queue.c)
target_link_libraries(fel_queue PRIVATE dvbridge_core)
target_link_options(fel_queue PRIVATE
  -Wl,--wrap=avcodec_open2 -Wl,--wrap=avcodec_send_packet
  -Wl,--wrap=avcodec_receive_frame -Wl,--wrap=avcodec_flush_buffers)
add_test(NAME fel_queue COMMAND fel_queue)
pkg_check_modules(DVGbm REQUIRED IMPORTED_TARGET gbm)
pkg_check_modules(DVDrm REQUIRED IMPORTED_TARGET libdrm)
add_library(drm_state_compile OBJECT "${DVBRIDGE_KODI_SOURCE}/xbmc/windowing/gbm/drm/DVBridgeState.cpp"
                                    "${DVBRIDGE_KODI_SOURCE}/xbmc/windowing/gbm/drm/DRMAtomic.cpp")
target_include_directories(drm_state_compile PRIVATE "${LAB}/tests/stubs" "${DVBRIDGE_KODI_SOURCE}/xbmc")
target_link_libraries(drm_state_compile PRIVATE PkgConfig::DVGbm PkgConfig::DVDrm)
add_executable(drm_atomic_transaction "${DVBRIDGE_KODI_SOURCE}/xbmc/windowing/gbm/drm/DRMAtomic.cpp"
                                     "${DVBRIDGE_KODI_SOURCE}/xbmc/windowing/gbm/drm/DVBridgeState.cpp"
                                     ../drm_atomic_transaction.cpp)
target_include_directories(drm_atomic_transaction PRIVATE "${LAB}/tests/stubs" "${DVBRIDGE_KODI_SOURCE}/xbmc")
target_link_libraries(drm_atomic_transaction PRIVATE PkgConfig::DVGbm PkgConfig::DVDrm)
add_test(NAME drm_atomic_transaction COMMAND drm_atomic_transaction)
add_executable(gbm_transaction "${DVBRIDGE_KODI_SOURCE}/xbmc/windowing/gbm/GBMUtils.cpp" ../gbm_transaction.cpp)
target_include_directories(gbm_transaction PRIVATE "${LAB}/tests/stubs" "${DVBRIDGE_KODI_SOURCE}/xbmc")
target_link_libraries(gbm_transaction PRIVATE PkgConfig::DVGbm)
add_test(NAME gbm_transaction COMMAND gbm_transaction)
add_test(NAME kodi_bridge_smoke COMMAND kodi_bridge_smoke)
set_tests_properties(kodi_bridge_smoke PROPERTIES ENVIRONMENT
  "LIBGL_ALWAYS_SOFTWARE=1;GALLIUM_DRIVER=llvmpipe;EGL_PLATFORM=surfaceless")
